# The LLM path goes through llm_client.get_client (anthropic | groq). Both
# imports are lazy, so the agent still runs (with no model calls) when neither
# a provider SDK nor an API key is present.
#   - Groq (TECHJAM_LLM_PROVIDER=groq): needs nothing here, it is raw urllib.
#   - Anthropic (the default when ANTHROPIC_API_KEY is set): needs this.
#
# Also used by the catalog normalisation pipeline:
#   python -m dataset_normalisation.pipeline run --no-llm   stdlib only
#   python -m dataset_normalisation.pipeline run            needs anthropic
#
# The major is capped deliberately: anthropic went 0.x -> 1.x with breaking
# changes to the batches/types surface that pipeline calls, so an uncapped
# install would silently break dataset_normalisation/columns/style_use_case.py.
anthropic>=1.2,<2
